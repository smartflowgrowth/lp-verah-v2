# LP Verah — Plano de Implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir a landing page do Verah como uma sessão única de scroll narrativo, em 8 blocos, com CTA persistente e formulário de lead validado.

**Architecture:** Next.js com export estático. Todo o conteúdo vive em `content/verah.ts` (tipado), e nenhum componente de bloco o conhece — recebem props. Animação dividida por responsabilidade: Motion para entradas discretas, GSAP/ScrollTrigger para scrub e pin, com um único contexto global. Lógica pura (progresso, telefone, schema) fica fora do React e é testada com Vitest.

**Tech Stack:** Next.js 16.3.4 · React 19.2.8 · TypeScript 7.0.2 · Tailwind 4.3.3 · motion 13.2.0 · gsap 3.15.0 · zod 4.5.4 · react-hook-form 7.87.0 · Vitest 5.0.0

### Três divergências deliberadas em relação ao spec

O spec descreve a arquitetura em traços largos; estes três pontos foram
resolvidos de outra forma na hora de escrever o código, e o plano vence:

1. **`useReducedMotion.ts` virou `useSessionMotion.ts`.** O nome original
   colidiria com o `useReducedMotion` do próprio `motion/react`, que o hook
   consome por dentro. Dois símbolos com o mesmo nome no mesmo arquivo é
   confusão garantida.
2. **`useBlockObserver.ts` não existe.** A observação de viewport é uma linha
   de `useInView` dentro de `Block.tsx`. Um arquivo separado para isso seria
   abstração vazia.
3. **Do shadcn, só o `drawer`.** Os campos do formulário são poucos e muito
   estilizados pela marca; `Input` e `Label` genéricos seriam uma camada a
   mais sem ganho. O `drawer` entra porque resolve acessibilidade real.

Criar `lib/progresso.ts`:

```ts
/**
 * Fração da sessão já percorrida, entre 0 e 1.
 * Pura de propósito: o bounce do iOS produz scrollY negativo e maior que o
 * total, e o trilho não pode passar dos limites por causa disso.
 */
export function calcularProgresso(
  scrollY: number,
  alturaDocumento: number,
  alturaJanela: number,
): number {
  const rolavel = alturaDocumento - alturaJanela
  if (rolavel <= 0) return 0
```

```ts
'use client'

import { useEffect, useState } from 'react'
import { calcularProgresso } from '@/lib/progresso'

/**
 * Um único ponto de leitura de scroll na aplicação inteira. O trilho, o CTA
 * fixo e o bloco ativo consomem daqui — não abrem observers próprios.
 */
export function useScrollProgress(): number {
  const [progresso, setProgresso] = useState(0)

  useEffect(() => {
    let quadro = 0

    const medir = () => {
      quadro = 0
      setProgresso(
        calcularProgresso(
          window.scrollY,
          document.documentElement.scrollHeight,
          window.innerHeight,
        ),
      )
    }

    const agendar = () => {
      if (quadro === 0) quadro = requestAnimationFrame(medir)
    }

    medir()
    window.addEventListener('scroll', agendar, { passive: true })
    window.addEventListener('resize', agendar)

    return () => {
      if (quadro !== 0) cancelAnimationFrame(quadro)
      window.removeEventListener('scroll', agendar)
      window.removeEventListener('resize', agendar)
    }
  }, [])

  return progresso
}
```

```ts
/**
 * Reduz o telefone aos 10 ou 11 dígitos que a Edge Function lead-lp aceita.
 * O código do país só é removido quando sobra um número válido sem ele —
 * 55 também é DDD do Rio Grande do Sul.
 */
export function normalizarTelefone(bruto: string): string {
  const digitos = bruto.replace(/\D/g, '')
  if (digitos.length > 11 && digitos.startsWith('55')) {
    return digitos.slice(2)
  }
  return digitos
}

/** Máscara progressiva para exibição durante a digitação. */
export function formatarTelefone(bruto: string): string {
  const d = normalizarTelefone(bruto).slice(0, 11)
  if (d.length === 0) return ''
  if (d.length <= 2) return `(${d}`
  const corte = d.length > 10 ? 7 : 6
  if (d.length <= corte) return `(${d.slice(0, 2)}) ${d.slice(2)}`
  return `(${d.slice(0, 2)}) ${d.slice(2, corte)}-${d.slice(corte)}`
}
```

```tsx
import { verah } from '@/content/verah'
import { useSessionMotion } from '@/lib/scroll/useSessionMotion'

/**
 * Sempre visível, exceto no bloco final — lá o formulário já está na tela e o
 * botão competiria consigo mesmo.

export function StickyCta() {

  const modo = useSessionMotion()
  const [pulsar, setPulsar] = useState(false)

  useEffect(() => {

      clearInterval(intervalo)
      clearTimeout(ocioso)
      window.removeEventListener('scroll', cancelar)

    <AnimatePresence>
      {!oculto && (
        <motion.div
          initial={{ opacity: 0, y: 16 }}
          animate={{ opacity: 1, y: 0 }}
          exit={{ opacity: 0, y: 16 }}
          transition={{ duration: modo.duracao * 0.6 }}
          className="fixed inset-x-0 bottom-0 z-50 px-4 pb-[max(1rem,env(safe-area-inset-bottom))] md:inset-x-auto md:right-8 md:bottom-8 md:px-0 md:pb-0"
```

```tsx
import { useEffect, useRef, useState } from 'react'
import gsap from 'gsap'
import { ScrollTrigger } from 'gsap/ScrollTrigger'
import { Block } from '../Block'
import { verah } from '@/content/verah'
import { useSessao } from '../SessionProvider'
import { useSessionMotion } from '@/lib/scroll/useSessionMotion'

gsap.registerPlugin(ScrollTrigger)

/**
 * Desktop: seção pinada, as quatro tipologias passam com scrub horizontal.
 * Mobile: rolagem horizontal por swipe, sem pin — pin em toque briga com o
 * gesto nativo e trava a página.
 */
export function Plantas() {
  const raiz = useRef<HTMLDivElement>(null)
  const trilho = useRef<HTMLDivElement>(null)
  const modo = useSessionMotion()
  const { abrirDrawer } = useSessao()
  const [visivel, setVisivel] = useState(verah.tipologias[0])

  useEffect(() => {
    if (!modo.anima || !raiz.current || !trilho.current) return
    if (!window.matchMedia('(min-width: 768px)').matches) return

    const ctx = gsap.context(() => {
      const percorrer = trilho.current!.scrollWidth - window.innerWidth
      if (percorrer <= 0) return

      gsap.to(trilho.current, {
        x: -percorrer,
        ease: 'none',
        scrollTrigger: {
          trigger: raiz.current,
          start: 'top top',
          end: () => `+=${percorrer}`,
          scrub: 0.8,
          pin: true,
          anticipatePin: 1,
          invalidateOnRefresh: true,
          onUpdate: (self) => {
            const i = Math.min(
              verah.tipologias.length - 1,
              Math.floor(self.progress * verah.tipologias.length),
            )
            setVisivel(verah.tipologias[i])
          },
        },
      })
    }, raiz)

    return () => ctx.revert()
  }, [modo.anima])

  return (
    <Block id="plantas" className="overflow-hidden">
      <div ref={raiz} className="relative">
        <div className="px-6 pt-28 md:pt-32">
```

Criar `components/session/blocks/Construtora.tsx`:

```tsx
'use client'

import { useEffect, useRef, useState } from 'react'
import { useInView } from 'motion/react'
import { Block } from '../Block'
import { verah } from '@/content/verah'
import { useSessionMotion } from '@/lib/scroll/useSessionMotion'

function Contador({
  alvo,
  prefixo,
  sufixo,
  formatar,
  anima,
}: {
  alvo: number
  prefixo: string
  sufixo: string
  formatar: boolean
  anima: boolean
}) {
  const ref = useRef<HTMLSpanElement>(null)
  const visivel = useInView(ref, { once: true, amount: 0.6 })
  const [valor, setValor] = useState(anima ? 0 : alvo)

  useEffect(() => {
    if (!visivel || !anima) return
    const inicio = performance.now()
    const duracao = 1800
    let quadro = requestAnimationFrame(function passo(agora) {
      const t = Math.min(1, (agora - inicio) / duracao)
      const suave = 1 - Math.pow(1 - t, 3)
      setValor(Math.round(alvo * suave))
      if (t < 1) quadro = requestAnimationFrame(passo)
    })
    return () => cancelAnimationFrame(quadro)
  }, [visivel, anima, alvo])

  const texto = formatar ? valor.toLocaleString('pt-BR') : String(valor)

  return (
    <span ref={ref}>
      {prefixo}
      {texto}
      {sufixo}
    </span>
  )
}

export function Construtora() {
  const modo = useSessionMotion()
  const { construtora } = verah

  return (
    <Block id="construtora" className="bg-cocoa px-6 py-28 text-cream md:py-40">
      <div className="mx-auto max-w-5xl text-center">
        <p className="text-[11px] uppercase tracking-[0.32em] opacity-60">{construtora.nome}</p>
        <h2 className="mt-6 font-display text-[clamp(2rem,6vw,4rem)] leading-[1.05]">
          {construtora.titulo}
        </h2>

        <dl className="mt-20 grid grid-cols-2 gap-12 md:grid-cols-4">
          {construtora.numeros.map((n) => (
            <div key={n.rotulo}>
              <dt className="font-display text-[clamp(1.75rem,4vw,3rem)] leading-none">
                <Contador
                  alvo={n.valor}
                  prefixo={n.prefixo}
                  sufixo={n.sufixo}
                  formatar={n.formatar}
                  anima={modo.anima}
                />
              </dt>
              <dd className="mt-3 text-[10px] uppercase tracking-[0.18em] opacity-60">
                {n.rotulo}
              </dd>
            </div>
          ))}
        </dl>
      </div>
    </Block>
  )
}
```
